input[type="text"]
	input - 輸入內容時觸發（即時）
	change - 內容改變且失去焦點時觸發
	blur - 失去焦點時觸發
	focus - 獲得焦點時觸發
	keydown - 按鍵按下時觸發
	keyup - 按鍵釋放時觸發
	
-----------------------------------------------------------------------------------------------

input[type="file"]
	change - 選擇檔案時觸發 <- 最重要!!!!!
	click - 點擊選擇檔案按鈕時觸發
	focus - 獲得焦點時觸發
	blur - 失去焦點時觸發
	
	注意：
	不支持 input 事件
	不能使用 v-model，必須用 @change
	
-----------------------------------------------------------------------------------------------

input[type="radio"]
	change - 選中狀態改變時觸發
	click - 點擊時觸發
	focus - 獲得焦點時觸發
	blur - 失去焦點時觸發
	
	注意：
	不支持 input 事件
	通常只用 change 事件配合 v-model

-----------------------------------------------------------------------------------------------

input[type="checkbox"]
	change - 勾選狀態改變時觸發
	click - 點擊時觸發
	focus - 獲得焦點時觸發
	blur - 失去焦點時觸發
	
	注意：
	不支持 input 事件
	通常只用 change 事件配合 v-model

-----------------------------------------------------------------------------------------------

select
	change - 選項改變時觸發
	focus - 獲得焦點時觸發
	blur - 失去焦點時觸發
	click - 點擊下拉選單時觸發
	
	注意：
	不支持 input 事件
	主要使用 change 事件

-----------------------------------------------------------------------------------------------

textarea
	click - 點擊時觸發
	input - 輸入內容時觸發（即時）
	change - 內容改變且失去焦點時觸發
	blur - 失去焦點時觸發
	focus - 獲得焦點時觸發
	keydown - 按鍵按下時觸發
	keyup - 按鍵釋放時觸發
	scroll - 捲動時觸發
	resize - 大小改變時觸發（如果可調整大小）
